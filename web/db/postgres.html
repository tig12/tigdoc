# drop all tables of schema (without deleting the schema)
DO $$ DECLARE
    r RECORD;
BEGIN
    FOR r IN (SELECT tablename FROM pg_tables WHERE schemaname = current_schema()) LOOP
        EXECUTE 'DROP TABLE IF EXISTS ' || quote_ident(r.tablename) || ' CASCADE';
    END LOOP;
END $$;

Access to fields
================
The operator -> returns JSON object field by key (returns json)
The operator ->> returns JSON object field by text (returns text)

# assoc array
select slug,ids_in_sources->'A1' from person;
select birth->'date-ut' from person limit 1;

# regular array
select history->0 from person limit 1;
select history->>0 from person limit 1;

select history->0->'source' from person limit 1;
select history->0->>'source' from person limit 1;
select * from person where history->0->>'source'='a1';
select * from person where ids_in_sources->>'a2'='2';
select birth->'place'->>'cy', count(*) from person
    group by birth->'place'->>'cy'
    order by count(*) desc;
--  order by birth->'place'->>'cy';

loop
================
select jsonb_array_elements(sources) from person limit 1;
select jsonb_each(ids_in_sources) from person limit 1;

Containment <@
================
# regular array
select count(*) from person where '["WR"]' <@ occus;
select count(*) from person where '"WR"' <@ occus;
select * from person where '"a4"' <@ sources and '"5muller_writers"' <@ sources and not '"a6"' <@ sources;
# associative array
select slug,occus,ids_in_sources from person where '{"1":"actor"}' <@ occus;

Existence ?
================
# where birth.place has a field geoid
select slug,birth from person where birth->'place' ? 'geoid' limit 1;

# where name doesn't have a field full
select count(*) from person where not(name ? 'full')

array_length
================
select slug,issues from person where jsonb_array_length(issues) != 0;

Arrays

Declaration

# Here, pay_by_quarter is a one-dimensional array
# and schedule is a two-dimensional array

CREATE TABLE sal_emp (
    name            text,
    pay_by_quarter  integer[],
    schedule        text[][]
);

# Also possible to specify the size of the array:

CREATE TABLE tictactoe (
    squares   integer[3][3]
);
WARNING : for postgres 18, dimension and size are not considered at runtime (same type) - useful for documentation.

Alternative syntax, ARRAY keyword: 
pay_by_quarter  integer ARRAY[4],

# or without dimension:
pay_by_quarter  integer ARRAY,

Array values

General form: '{ val1 delim val2 delim ... }', where delim = , (coma), except for type box (delim is ;).
val can be a litteral value, a sub-array or NULL.
Alternative syntax: 'ARRAY[val1 delim val2 ... ]'
Ex: 
'{{1,2,3},{4,5,6},{7,8,9}}'

# Alternative syntax:
ARRAY[['meeting', 'lunch'], ['training', 'presentation']]
By default, indices start with 1.
But indices are not necessarily 1-based: myarray[-2:7]

Array slices

General syntax is lower-bound:upper-bound 
SELECT schedule[1:2][1:1] FROM sal_emp WHERE name = 'Bill';

# Lower bound or upper bound can be omitted
SELECT schedule[:2][2:] FROM sal_emp WHERE name = 'Bill';
SELECT schedule[:][1:1] FROM sal_emp WHERE name = 'Bill';
If one dimension is declared as a slice, all dimensions are treated as slices.
In this context, a single number is interpreted as a slice : 3 is interpreted as [1:3] => don't use to avoid confusion.

Insert

INSERT INTO sal_emp
    VALUES ('Bill',
    '{10000, 10000, 10000, 10000}',
    '{{"meeting", "lunch"}, {"training", "presentation"}}');

# Alternative syntax:
INSERT INTO sal_emp
    VALUES ('Bill',
    ARRAY[10000, 10000, 10000, 10000],
    ARRAY[['meeting', 'lunch'], ['training', 'presentation']]);

Select

Out of bound indices or array reference with the wrong number of subscripts yields a null rather than an error. 
SELECT name FROM sal_emp WHERE pay_by_quarter[1] <> pay_by_quarter[2];
SELECT pay_by_quarter[3] FROM sal_emp;

select id,nom from acteur where id = any('{"3", "2"}');
# Alternative syntax:
select id,nom from acteur where id = any(array[2, 3]);
Array slices: 
SELECT schedule[1:2][1:1] FROM sal_emp WHERE name = 'Bill';

# Omit lower or upper bound:
SELECT schedule[:2][2:] FROM sal_emp WHERE name = 'Bill';
If any dimension is written as a slice (contains a colon), then all dimensions are treated as slices. Any dimension that has only a single number (no colon) is treated as being from 1 to the number specified. For example, [2] is treated as [1:2] => avoid, specify slices for all dimension to avoid confusion

Update

UPDATE sal_emp SET pay_by_quarter = '{25000,25000,27000,27000}' WHERE name = 'Carol';
UPDATE sal_emp SET pay_by_quarter = ARRAY[25000,25000,27000,27000] WHERE name = 'Carol';

UPDATE sal_emp SET pay_by_quarter[4] = 15000 WHERE name = 'Bill';

# Update a slice:
UPDATE sal_emp SET pay_by_quarter[1:2] = '{27000,27000}' WHERE name = 'Carol';

https://postgrest.org/en/stable/api.html

    /todos
    /groop?type=eq.history
    /people?age=gte.18&student=is.true
    /people?or=(age.lt.18,age.gt.21)
    /people?grade=gte.90&student=is.true&or=(age.eq.14,not.and(age.gte.11,age.lte.17))
You can request table/columns with spaces in them by percent encoding the spaces with %20:
    /Order%20Items?Unit%20Price=lt.200 
If filters include PostgREST reserved characters(,, ., :, ()) you’ll have to surround them in percent encoded double quotes %22 for correct processing.
    /employees?name=in.(%22Hebdon,John%22,%22Williams,Mary%22)
    /people?order=age.desc,height.asc
If no direction is specified it defaults to ascending order:
    /people?order=age
    /people?limit=15&offset=30
