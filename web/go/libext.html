goph - ssh client

import (
	"github.com/melbahja/goph"
)

auth, err := goph.UseAgent()
client, err := goph.NewConn(&goph.Config{
    User:     user,
    Addr:     host,
    Port:     22,
    Auth:     auth,
    Timeout:  goph.DefaultTimeout,
    Callback: ssh.InsecureIgnoreHostKey(),
})
defer client.Close()
sftp, err := client.NewSftp()
